class AppState {

    constructor() {

        this.testData = null; // meta, passage, questions, etc.

        this.attempt = {

            testId: null,

            answers: {}, // questionId -> value

            flags: [],

            timer: 0,

            status: 'idle', // idle, running, paused, finished

            tabSwitches: 0,

            switchLog: [],

            highlights: [], // {line, start, end, class}

            changeLog: [] // {ts, qId, val}

        };

        this.prefs = {

            splitRatio: 50,

            fontSize: 'M'

        };

    }


    loadFromStorage(testId) {

        const key = `ielts_attempt_${testId}`;

        const saved = localStorage.getItem(key);

        if (saved) {

            this.attempt = JSON.parse(saved);

        } else {

            this.attempt.testId = testId;

        }

        

        // Load prefs

        const prefs = localStorage.getItem('ielts_prefs');

        if (prefs) this.prefs = JSON.parse(prefs);

    }


    saveToStorage() {

        if (!this.attempt.testId) return;

        localStorage.setItem(`ielts_attempt_${this.attempt.testId}`, JSON.stringify(this.attempt));

        localStorage.setItem('ielts_prefs', JSON.stringify(this.prefs));

    }


    logChange(qId, val) {

        this.attempt.changeLog.push({

            ts: Date.now(),

            qId: qId,

            val: val

        });

        this.saveToStorage();

    }


    exportAttempt() {

        const dataStr = "data:text/json;charset=utf-8," + encodeURIComponent(JSON.stringify(this.attempt));

        const downloadAnchorNode = document.createElement('a');

        downloadAnchorNode.setAttribute("href", dataStr);

        downloadAnchorNode.setAttribute("download", `attempt_${this.attempt.testId}_${Date.now()}.json`);

        document.body.appendChild(downloadAnchorNode);

        downloadAnchorNode.click();

        downloadAnchorNode.remove();

    }


    importAttempt(input) {

        const file = input.files[0];

        if (!file) return;

        const reader = new FileReader();

        reader.onload = (e) => {

            try {

                const data = JSON.parse(e.target.result);

                if(data.testId !== this.testData.meta.testId) {

                    alert("Test ID mismatch!"); 

                    return;

                }

                this.attempt = data;

                this.saveToStorage();

                window.location.reload();

            } catch(err) { alert("Invalid JSON"); }

        };

        reader.readAsText(file);

    }

    

    resetAttempt() {

        if(confirm("Are you sure? This will delete all progress.")) {

            localStorage.removeItem(`ielts_attempt_${this.attempt.testId}`);

            window.location.reload();

        }

    }

}


class CSVParser {

    constructor(app) { this.app = app; }


    parseCSV(text) {

        // Simple robust parser handling quoted fields

        const rows = [];

        let currentRow = [];

        let currentField = '';

        let insideQuotes = false;

        

        for (let i = 0; i < text.length; i++) {

            const char = text[i];

            const nextChar = text[i+1];

            

            if (char === '"') {

                if (insideQuotes && nextChar === '"') {

                    currentField += '"'; i++; // Skip escaped quote

                } else {

                    insideQuotes = !insideQuotes;

                }

            } else if (char === ',' && !insideQuotes) {

                currentRow.push(currentField.trim()); currentField = '';

            } else if ((char === '\r' || char === '\n') && !insideQuotes) {

                if (currentField || currentRow.length > 0) currentRow.push(currentField.trim());

                if (currentRow.length > 0) rows.push(currentRow);

                currentRow = []; currentField = '';

                if (char === '\r' && nextChar === '\n') i++;

            } else {

                currentField += char;

            }

        }

        if (currentField || currentRow.length > 0) {

            currentRow.push(currentField.trim());

            rows.push(currentRow);

        }

        return rows;

    }


    rowToObject(headers, row) {

        const obj = {};

        headers.forEach((h, i) => obj[h] = row[i] || '');

        return obj;

    }


    async processFiles() {

        const input = document.getElementById('csvInput');

        const files = Array.from(input.files);

        const log = document.getElementById('importLog');

        log.innerHTML = '';

        

        const rawData = {};

        const requiredFiles = ['meta', 'passage_lines', 'questions'];

        

        const readProms = files.map(f => new Promise(resolve => {

            const reader = new FileReader();

            reader.onload = (e) => {

                // Guess type by filename

                let type = null;

                const name = f.name.toLowerCase();

                if (name.includes('meta')) type = 'meta';

                else if (name.includes('passage')) type = 'passage_lines';

                else if (name.includes('questions')) type = 'questions';

                else if (name.includes('segment')) type = 'segments';

                else if (name.includes('pool')) type = 'pool';

                else if (name.includes('targets')) type = 'targets';

                else if (name.includes('statement')) type = 'statements';

                

                if(type) {

                    const rows = this.parseCSV(e.target.result);

                    const headers = rows[0]; // Assume first row is header

                    rawData[type] = rows.slice(1).map(r => this.rowToObject(headers, r));

                    log.innerHTML += `<div class="text-green">Loaded ${type} (${rawData[type].length} rows)</div>`;

                }

                resolve();

            };

            reader.readAsText(f);

        }));


        await Promise.all(readProms);


        // Validation

        const missing = requiredFiles.filter(k => !rawData[k]);

        if (missing.length > 0) {

            log.innerHTML += `<div class="text-red">Missing files: ${missing.join(', ')}</div>`;

            return;

        }


        this.compileTest(rawData);

    }


    compileTest(raw) {

        // Transform raw CSV data into App Object Model

        const test = {

            meta: raw.meta[0], // First row of meta

            passage: raw.passage_lines.map(l => ({ line: parseInt(l.lineNo), text: l.text })),

            questions: [],

            segments: raw.segments || [],

            headingsPool: raw.pool || [],

            targets: raw.targets || [],

            statements: raw.statements || []

        };


        // Enrich questions

        test.questions = raw.questions.map(q => {

            q.order = parseInt(q.order) || 999;

            // Attach relational data

            if (q.type === 'matching_headings') {

                const target = test.targets.find(t => t.questionId === q.questionId);

                if (target) {

                    q.targetSegment = target.segmentId;

                    q.correctHeading = target.correctHeadingKey;

                    q.poolId = target.poolId;

                }

            } else if (q.type === 'tfng' || q.type === 'ynng') {

                const stmt = test.statements.find(s => s.questionId === q.questionId);

                if (stmt) {

                    q.prompt = stmt.statementText;

                    q.correctLabel = stmt.correctLabel;

                }

            }

            return q;

        }).sort((a,b) => a.order - b.order);


        console.log("Compiled Test:", test);

        this.app.state.testData = test;

        this.app.state.attempt.testId = test.meta.testId;

        this.app.state.saveToStorage();

        this.app.ui.init();

        this.app.ui.closeModal('importModal');

    }

        // Group logic could go here, treating flat list for now

            if (input) {

                } else {

    renderHeadingPool(poolId) {

        const pool = this.app.state.testData.headingsPool.filter(p => p.poolId === poolId);

        if (pool.length === 0) return '';

        

        let tokens = pool.map(h => 

            `<div class="draggable-token" draggable="true" data-key="${h.headingKey}" data-pool="${poolId}" id="token-${h.headingKey}">

                <b>${h.headingKey}</b> ${h.headingText}

            </div>`

        ).join('');

        

        return `<div class="dd-container">

            <h4>List of Headings</h4>

            <div class="pool-area" id="pool-${poolId}" ondrop="app.logic.drop(event)" ondragover="app.logic.allowDrop(event)">

                ${tokens}

            </div>

        </div>`;

    }


    renderQuestionInput(q) {

        if (q.type === 'mcq_single') {

            const opts = ['optionA','optionB','optionC','optionD'].map(k => {

                if(!q[k]) return '';

                const key = k.replace('option','');

                return `<label class="mcq-option">

                    <input type="radio" name="ans-${q.questionId}" value="${key}" onchange="app.logic.saveAnswer('${q.questionId}', this.value)"> 

                    <b>${key}.</b> ${q[k]}

                </label>`;

            }).join('');

            return opts;

        } 

        else if (q.type === 'fill_blank') {

            return `<input type="text" class="input-blank" name="ans-${q.questionId}" onchange="app.logic.saveAnswer('${q.questionId}', this.value)">`;

        }

        else if (q.type === 'tfng' || q.type === 'ynng') {

            const labels = q.type === 'tfng' ? ['TRUE','FALSE','NOT GIVEN'] : ['YES','NO','NOT GIVEN'];

            const vals = q.type === 'tfng' ? ['T','F','NG'] : ['Y','N','NG'];

            return `<div class="tfng-group">

                ${vals.map((v, i) => 

                    `<div class="tfng-btn" data-qid="${q.questionId}" data-val="${v}" onclick="app.logic.selectTFNG(this)">${labels[i]}</div>`

                ).join('')}

            </div>`;

        }

        else if (q.type === 'matching_headings') {

            return `<div class="dd-zone" data-qid="${q.questionId}" ondrop="app.logic.drop(event)" ondragover="app.logic.allowDrop(event)">

                <span style="color:#ccc; font-size:0.8em; margin-right: auto;">Drop heading here</span>

                <!-- Fallback Select -->

                <select onchange="app.logic.saveAnswer('${q.questionId}', this.value)" onclick="event.stopPropagation()">

                    <option value="">(Select)</option>

                    ${this.app.state.testData.headingsPool.filter(h => h.poolId === q.poolId).map(h => `<option value="${h.headingKey}">${h.headingKey}</option>`).join('')}

                </select>

            </div>`;

        }

        return '(Unknown type)';

    }


    sanitize(str) {

        if(!str) return '';

        // Very basic sanitization allowing only safe tags

        return str.replace(/</g, "&lt;").replace(/>/g, "&gt;")

            .replace(/&lt;(b|i|u|sup|sub|br)&gt;/g, "<$1>")

            .replace(/&lt;\/(b|i|u|sup|sub)&gt;/g, "</$1>");

    }

}


class Logic {

    constructor(app) {

        this.app = app;

        this.timerInterval = null;

    }


    startTimer() {

        if (this.app.state.attempt.status === 'finished') return;

        this.app.state.attempt.status = 'running';

        const isCountdown = this.app.state.testData.meta.timerMode === 'countdown';

        const duration = parseInt(this.app.state.testData.meta.durationMinutes) * 60;

        

        if (this.app.state.attempt.timer === 0 && isCountdown) {

            this.app.state.attempt.timer = duration;

        }


        this.timerInterval = setInterval(() => {

            if (this.app.state.attempt.status !== 'running') return;

            

            if (isCountdown) {

                this.app.state.attempt.timer--;

                if (this.app.state.attempt.timer <= 0) this.submitTest();

            } else {

                this.app.state.attempt.timer++;

            }

            this.app.ui.updateTimerDisplay();

            if (this.app.state.attempt.timer % 10 === 0) this.app.state.saveToStorage(); // Autosave every 10s

        }, 1000);

    }


    toggleTimer() {

        const status = this.app.state.attempt.status;

        const btn = document.getElementById('btnPause');

        if (status === 'running') {

            this.app.state.attempt.status = 'paused';

            btn.innerText = "Resume";

            document.body.classList.add('blurred'); // Could add CSS blur filter

        } else if (status === 'paused' || status === 'idle') {

            this.startTimer();

            btn.innerText = "Pause";

        }

    }


    saveAnswer(qId, value) {

        if (this.app.state.attempt.status === 'finished') return;

        this.app.state.attempt.answers[qId] = value;

        this.app.state.logChange(qId, value);

        

        // UI Feedback

        document.querySelector(`#qblock-${qId} .q-id`).classList.add('answered');

    }


    selectTFNG(el) {

        if (this.app.state.attempt.status === 'finished') return;

        const qId = el.dataset.qid;

        const val = el.dataset.val;

        

        // Toggle visual

        const group = el.parentElement;

        group.querySelectorAll('.tfng-btn').forEach(b => b.classList.remove('selected'));

        el.classList.add('selected');

        

        this.saveAnswer(qId, val);

    }


    // --- Drag & Drop ---

    allowDrop(ev) { ev.preventDefault(); }

    

    // Add event listeners to tokens when rendered

    initDragEvents() {

        document.querySelectorAll('.draggable-token').forEach(t => {

            t.addEventListener('dragstart', (ev) => {

                ev.dataTransfer.setData("text/plain", JSON.stringify({

                    key: t.dataset.key, 

                    pool: t.dataset.pool,

                    id: t.id

                }));

            });

        });

    }


    drop(ev) {

        ev.preventDefault();

        const dataStr = ev.dataTransfer.getData("text");

        if (!dataStr) return;

        const data = JSON.parse(dataStr);

        

        // Determine target

        let target = ev.target;

        while (!target.classList.contains('dd-zone') && !target.classList.contains('pool-area')) {

            target = target.parentElement;

            if(!target) return;

        }


        const tokenEl = document.getElementById(data.id);

        

        if (target.classList.contains('dd-zone')) {

            // Drop into question

            // If zone already has a token, move it back to pool

            const existing = target.querySelector('.draggable-token');

            if (existing) {

                const pool = document.getElementById(`pool-${existing.dataset.pool}`);

                pool.appendChild(existing);

            }

            target.appendChild(tokenEl);

            const qId = target.dataset.qid;

            this.saveAnswer(qId, data.key);

        } else if (target.classList.contains('pool-area')) {

            // Return to pool

            target.appendChild(tokenEl);

            // Need to find which question held this token to clear answer

            // This is tricky, simpler to re-scan or just rely on user interaction

            // For now, if dragged to pool, we don't know easily which Q it came from without tracking

            // Hack: search all answers for this value and clear? No, value might duplicate.

            // Correct way: When dropped in zone, we update. When removed from zone, we update.

        }

    }

    

    restoreDragDrop() {

        this.initDragEvents();

        const answers = this.app.state.attempt.answers;

        Object.keys(answers).forEach(qid => {

            const val = answers[qid];

            const zone = document.querySelector(`.dd-zone[data-qid="${qid}"]`);

            if (zone) {

                // Find token in pool

                const token = document.querySelector(`.draggable-token[data-key="${val}"]`);

                if (token) zone.appendChild(token);

            }

        });

    }


    // --- Scoring ---

    submitTest() {

        this.app.state.attempt.status = 'finished';

        clearInterval(this.timerInterval);

        

        let correctCount = 0;

        let total = 0;

        let detailsHtml = '<div style="max-height: 400px; overflow: auto;">';


        this.app.state.testData.questions.forEach(q => {

            total++;

            const userAns = (this.app.state.attempt.answers[q.questionId] || '').trim();

            let isCorrect = false;

            let correctStr = '';


            if (['mcq_single', 'matching_headings', 'tfng', 'ynng'].includes(q.type)) {

                // Key comparison

                const key = q.correctOptionKey || q.correctHeading || q.correctLabel;

                isCorrect = userAns === key;

                correctStr = key;

            } else if (q.type === 'fill_blank') {

                const keys = q.correctAnswers.split('|').map(k => k.trim());

                if (q.caseSensitive === 'false' || !q.caseSensitive) {

                    isCorrect = keys.some(k => k.toLowerCase() === userAns.toLowerCase());

                } else {

                    isCorrect = keys.includes(userAns);

                }

                correctStr = keys.join(' OR ');

            }


            if (isCorrect) correctCount++;


            detailsHtml += `<div class="result-row">

                <div>

                    <b>${q.questionId}</b> (${q.type}) <br>

                    <span class="${isCorrect ? 'text-green' : 'text-red'}">

                        Your: ${userAns || '(empty)'}

                    </span>

                </div>

                <div class="text-right">

                    Expected: <b>${correctStr}</b>

                </div>

            </div>`;

        });


        detailsHtml += '</div>';


        const score = document.getElementById('scoreHeader');

        score.innerHTML = `Score: ${correctCount} / ${total} (${Math.round(correctCount/total*100)}%)`;

        document.getElementById('resultDetails').innerHTML = detailsHtml;

        this.app.ui.openModal('resultModal');

        this.app.state.saveToStorage();

    }

}
